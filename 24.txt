    <script>
        class Game24 {
            constructor() {
                this.numbers = [];
                this.expression = [];
                this.currentResult = 0;
                this.calculationSteps = [];
                this.usedNumbers = [];
                this.solvedCount = 0;
                this.attemptCount = 0;
                this.startTime = Date.now();
                this.timerInterval = null;
                this.lastOperator = null;
                this.waitingForNumber = true;
                this.calculationStack = []; // 计算栈
                this.currentCalculation = null; // 当前计算单元
                this.availableCards = []; // 可用卡牌数组
                this.selectedCards = []; // 已选择的卡牌
                this.currentOperator = null; // 当前运算符
                
                this.init();
            }
            
            init() {
                this.generateNumbers();
                this.renderNumbers();
                this.startTimer();
            }
            
            generateNumbers() {
                // 生成4个1-13的数字（代表扑克牌）
                this.numbers = [];
                for (let i = 0; i < 4; i++) {
                    this.numbers.push(Math.floor(Math.random() * 13) + 1);
                }
                this.selectedNumbers = [];
                this.usedNumbers = [];
                this.expression = [];
                this.availableCards = [...this.numbers]; // 初始化可用卡牌
                this.selectedCards = [];
                this.updateExpression();
            }
            
            renderNumbers() {
                const container = document.getElementById('numbersContainer');
                container.innerHTML = '';
                
                this.availableCards.forEach((num, index) => {
                    const card = document.createElement('div');
                    card.className = 'number-card';
                    card.textContent = num;
                    card.onclick = () => this.selectCard(index);
                    container.appendChild(card);
                });
            }
            
            selectCard(index) {
                const num = this.availableCards[index];
                const card = document.querySelectorAll('.number-card')[index];
                
                // 如果已经选择了2个卡牌，不能再选择
                if (this.selectedCards.length >= 2) {
                    this.showResult('已经选择了2个卡牌，请先计算！', 'error');
                    return;
                }
                
                // 添加卡牌到已选择列表
                this.selectedCards.push({
                    index: index,
                    value: num
                });
                
                // 视觉反馈
                card.classList.add('selected');
                
                // 如果选择了第一个数字，提示选择运算符
                if (this.selectedCards.length === 1) {
                    this.showResult('请选择运算符', 'info');
                }
                // 如果选择了第二个数字，自动执行计算
                else if (this.selectedCards.length === 2) {
                    this.executeCardCalculation();
                }
            }
            
            performCalculation() {
                if (this.selectedCards.length !== 2) {
                    return;
                }
                
                // 等待用户选择运算符
                this.showResult('请选择运算符进行计算', 'info');
            }
            
            handleNumberSelection(num) {
                if (this.currentCalculation === null) {
                    // 开始新的计算
                    this.currentCalculation = {
                        left: num,
                        operator: null,
                        right: null,
                        result: num
                    };
                    this.expression = [num];
                } else if (this.currentCalculation.operator === null) {
                    // 当前计算还没有运算符，替换左操作数
                    this.currentCalculation.left = num;
                    this.currentCalculation.result = num;
                    this.expression = [num];
                } else if (this.currentCalculation.right === null) {
                    // 当前计算有运算符但没有右操作数
                    this.currentCalculation.right = num;
                    this.expression = [this.currentCalculation.left, this.currentCalculation.operator, num];
                    // 执行计算
                    this.executeCurrentCalculation();
                    // 计算完成后，结果替换到第二个数字位置，继续计算
                    this.expression = [this.currentCalculation.result];
                    this.currentCalculation.left = this.currentCalculation.result;
                    this.currentCalculation.right = null;
                    this.currentCalculation.operator = null;
                } else {
                    // 当前计算已完成，开始新的计算
                    this.finalizeCurrentCalculation();
                    this.currentCalculation = {
                        left: num,
                        operator: null,
                        right: null,
                        result: num
                    };
                    this.expression = [num];
                }
                
                this.updateDisplay();
            }
            
            addOperator(op) {
                // 只处理基本运算符
                if (!['+', '-', '×', '÷'].includes(op)) {
                    return;
                }
                
                // 如果选择了1个数字，设置运算符
                if (this.selectedCards.length === 1) {
                    this.currentOperator = op;
                    this.showResult('请选择第二个数字', 'info');
                } else if (this.selectedCards.length === 0) {
                    this.showResult('请先选择第一个数字', 'error');
                } else {
                    this.showResult('已经选择了2个数字，请重新开始', 'error');
                }
            }
            
            executeCardCalculation() {
                if (this.selectedCards.length !== 2 || !this.currentOperator) {
                    return;
                }
                
                const left = this.selectedCards[0].value;
                const right = this.selectedCards[1].value;
                const op = this.currentOperator;
                
                // 计算结果
                let result;
                try {
                    switch (op) {
                        case '+':
                            result = left + right;
                            break;
                        case '-':
                            result = left - right;
                            break;
                        case '×':
                            result = left * right;
                            break;
                        case '÷':
                            if (right === 0) {
                                result = 'Error';
                            } else {
                                result = left / right;
                            }
                            break;
                        default:
                            result = left;
                    }
                } catch (error) {
                    result = 'Error';
                }
                
                // 记录计算步骤
                const step = `${left} ${op} ${right} = ${result}`;
                this.calculationStack.push({
                    step: step,
                    left: left,
                    operator: op,
                    right: right,
                    result: result
                });
                
                // 更新可用卡牌：移除选中的两个卡牌，添加结果卡牌
                this.availableCards = this.availableCards.filter((card, index) => 
                    !this.selectedCards.some(selected => selected.index === index)
                );
                this.availableCards.push(result);
                
                // 清空选中状态和运算符
                this.selectedCards = [];
                this.currentOperator = null;
                
                // 更新显示
                this.renderNumbers();
                this.updateResultDisplay();
                this.hideResult();
                
                // 检查是否完成
                if (this.availableCards.length === 1) {
                    this.checkFinalResult();
                } else {
                    this.showResult('请继续选择两个数字进行计算', 'info');
                }
            }
            
            checkFinalResult() {
                const finalResult = this.availableCards[0];
                if (finalResult === 24) {
                    this.solvedCount++;
                    this.updateScore();
                    this.showResult('恭喜！答案正确！', 'success');
                    setTimeout(() => this.newGame(), 2000);
                } else {
                    this.updateScore();
                    this.showResult(`最终结果: ${finalResult}，继续努力！`, 'info');
                }
            }
            
            executeCurrentCalculation() {
                if (this.currentCalculation.left !== null && 
                    this.currentCalculation.operator !== null && 
                    this.currentCalculation.right !== null) {
                    
                    try {
                        const left = this.currentCalculation.left;
                        const right = this.currentCalculation.right;
                        const operator = this.currentCalculation.operator;
                        
                        let result;
                        switch (operator) {
                            case '+':
                                result = left + right;
                                break;
                            case '-':
                                result = left - right;
                                break;
                            case '×':
                                result = left * right;
                                break;
                            case '÷':
                                if (right === 0) {
                                    result = 'Error';
                                } else {
                                    result = left / right;
                                }
                                break;
                            default:
                                result = left;
                        }
                        
                        this.currentCalculation.result = result;
                        this.currentResult = result;
                        
                        // 更新计算步骤 - 显示完整的计算过程
                        const step = `${left} ${operator} ${right} = ${result}`;
                        this.calculationSteps = [step];
                        
                        // 将步骤添加到计算栈中
                        this.calculationStack.push({
                            step: step,
                            left: left,
                            operator: operator,
                            right: right,
                            result: result
                        });
                        
                    } catch (error) {
                        this.currentCalculation.result = 'Error';
                        this.currentResult = 'Error';
                        this.calculationSteps = ['计算错误'];
                    }
                }
            }
            
            finalizeCurrentCalculation() {
                // 计算步骤已经在executeCurrentCalculation中添加了，这里不需要重复添加
                // 这个方法保留用于其他可能的清理工作
            }
            
            updateDisplay() {
                this.updateExpression();
                this.updateResultDisplay();
            }
            
            updateExpression() {
                const expressionDiv = document.getElementById('expression');
                if (this.expression.length === 0) {
                    expressionDiv.innerHTML = '<span style="color: #999;">点击数字和运算符开始计算...</span>';
                    return;
                }
                
                expressionDiv.innerHTML = this.expression.map(item => 
                    `<span class="expression-item">${item}</span>`
                ).join('');
            }
            
            
            updateResultDisplay() {
                document.getElementById('currentResult').textContent = this.currentResult;
                
                const stepsDiv = document.getElementById('calculationSteps');
                // 显示所有计算步骤
                const allSteps = this.calculationStack.map(item => item.step);
                if (allSteps.length > 0) {
                    stepsDiv.innerHTML = allSteps.map(step => 
                        `<span class="step-item">${step}</span>`
                    ).join('');
                } else if (this.calculationSteps.length > 0) {
                    stepsDiv.innerHTML = this.calculationSteps.map(step => 
                        `<span class="step-item">${step}</span>`
                    ).join('');
                } else {
                    stepsDiv.innerHTML = '';
                }
            }
            
            submitAnswer() {
                // 检查是否还有卡牌可以计算
                if (this.availableCards.length > 1) {
                    this.showResult('还有卡牌可以计算，请继续！', 'error');
                    return;
                }
                
                // 检查最终结果
                if (this.availableCards.length === 1) {
                    this.attemptCount++;
                    this.checkFinalResult();
                } else {
                    this.showResult('请先进行计算！', 'error');
                }
            }
            
            clearExpression() {
                this.expression = [];
                this.usedNumbers = [];
                this.currentResult = 0;
                this.calculationSteps = [];
                this.calculationStack = [];
                this.currentCalculation = null;
                this.lastOperator = null;
                this.waitingForNumber = true;
                this.availableCards = [...this.numbers];
                this.selectedCards = [];
                this.currentOperator = null;
                
                document.querySelectorAll('.number-card').forEach(card => {
                    card.classList.remove('selected');
                });
                
                this.renderNumbers();
                this.updateDisplay();
                this.hideResult();
                this.hideAnswer();
            }
            
            newGame() {
                this.generateNumbers();
                this.renderNumbers();
                this.clearExpression();
                this.startTime = Date.now();
                this.hideResult();
                this.hideAnswer();
            }
            
            hideAnswer() {
                document.getElementById('answerDisplay').style.display = 'none';
            }
            
            showResult(message, type) {
                const resultDiv = document.getElementById('result');
                resultDiv.textContent = message;
                resultDiv.className = `result ${type}`;
                resultDiv.style.display = 'block';
            }
            
            hideResult() {
                document.getElementById('result').style.display = 'none';
            }
            
            updateScore() {
                document.getElementById('solvedCount').textContent = this.solvedCount;
                document.getElementById('attemptCount').textContent = this.attemptCount;
                const successRate = this.attemptCount > 0 ? Math.round((this.solvedCount / this.attemptCount) * 100) : 0;
                document.getElementById('successRate').textContent = successRate + '%';
            }
            
            // 24点求解算法
            solve24() {
                const solutions = [];
                const nums = [...this.numbers];
                const operators = ['+', '-', '*', '/'];
                
                // 生成所有数字的排列
                const permutations = this.getPermutations(nums);
                
                // 生成所有运算符组合
                const operatorCombinations = this.getOperatorCombinations(3);
                
                // 生成所有括号位置
                const bracketPatterns = [
                    '((a op1 b) op2 c) op3 d',
                    '(a op1 (b op2 c)) op3 d',
                    'a op1 ((b op2 c) op3 d)',
                    'a op1 (b op2 (c op3 d))',
                    '(a op1 b) op2 (c op3 d)'
                ];
                
                for (let perm of permutations) {
                    for (let ops of operatorCombinations) {
                        for (let pattern of bracketPatterns) {
                            try {
                                const expr = this.buildExpression(pattern, perm, ops);
                                const result = eval(expr);
                                
                                if (Math.abs(result - 24) < 0.0001) {
                                    const displayExpr = expr.replace(/\*/g, '×').replace(/\//g, '÷');
                                    if (!solutions.includes(displayExpr)) {
                                        solutions.push(displayExpr);
                                    }
                                }
                            } catch (e) {
                                // 忽略除零等错误
                            }
                        }
                    }
                }
                
                return solutions;
            }
            
            getPermutations(arr) {
                if (arr.length <= 1) return [arr];
                const result = [];
                for (let i = 0; i < arr.length; i++) {
                    const current = arr[i];
                    const remaining = arr.slice(0, i).concat(arr.slice(i + 1));
                    const perms = this.getPermutations(remaining);
                    for (let perm of perms) {
                        result.push([current, ...perm]);
                    }
                }
                return result;
            }
            
            getOperatorCombinations(length) {
                const operators = ['+', '-', '*', '/'];
                const result = [];
                
                function generate(current) {
                    if (current.length === length) {
                        result.push([...current]);
                        return;
                    }
                    for (let op of operators) {
                        current.push(op);
                        generate(current);
                        current.pop();
                    }
                }
                
                generate([]);
                return result;
            }
            
            buildExpression(pattern, numbers, operators) {
                let expr = pattern;
                expr = expr.replace(/a/g, numbers[0]);
                expr = expr.replace(/b/g, numbers[1]);
                expr = expr.replace(/c/g, numbers[2]);
                expr = expr.replace(/d/g, numbers[3]);
                expr = expr.replace(/op1/g, operators[0]);
                expr = expr.replace(/op2/g, operators[1]);
                expr = expr.replace(/op3/g, operators[2]);
                return expr;
            }
            
            showAnswer() {
                const answerDisplay = document.getElementById('answerDisplay');
                const answerList = document.getElementById('answerList');
                
                if (answerDisplay.style.display === 'none' || answerDisplay.style.display === '') {
                    // 显示答案
                    const solutions = this.solve24();
                    
                    if (solutions.length === 0) {
                        answerList.innerHTML = '<div class="no-answer">这组数字无法算出24点！</div>';
                    } else {
                        answerList.innerHTML = solutions.map((solution, index) => 
                            `<div class="answer-item">${index + 1}. ${solution}</div>`
                        ).join('');
                    }
                    
                    answerDisplay.style.display = 'block';
                } else {
                    // 隐藏答案
                    answerDisplay.style.display = 'none';
                }
            }
            
            startTimer() {
                this.timerInterval = setInterval(() => {
                    const elapsed = Math.floor((Date.now() - this.startTime) / 1000);
                    const minutes = Math.floor(elapsed / 60);
                    const seconds = elapsed % 60;
                    document.getElementById('timeDisplay').textContent = 
                        `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
                }, 1000);
            }
